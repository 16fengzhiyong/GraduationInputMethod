# 交互设计规范

## 1. 设计理念

### 1.1 核心目标
打造**直觉、高效、愉悦**的交互体验，让用户的每一次操作都能得到即时、准确、舒适的反馈。

### 1.2 设计原则
- **直觉性**：操作符合用户预期
- **高效性**：最少步骤完成任务
- **反馈性**：每个操作都有响应
- **容错性**：允许误操作并提供恢复

## 2. 手势交互

### 2.1 基础手势

| 手势 | 触发区域 | 响应 | 用途 |
|------|----------|------|------|
| **点击** | 按键区域 | 输入字符 | 基础输入 |
| **长按** | 按键区域 | 弹出备选字符 | 输入特殊字符 |
| **滑动上** | 按键区域 | 输入数字/符号 | 快速输入数字 |
| **滑动左** | 候选词 | 删除候选词 | 删除自造词 |
| **滑动右** | 候选词 | 置顶候选词 | 常用词置顶 |
| **双指缩放** | 键盘区域 | 调整键盘大小 | 自定义大小 |

### 2.2 手势识别逻辑

```kotlin
// 手势识别器
class GestureRecognizer {
    // 手势阈值
    companion object {
        const val TAP_TIMEOUT = 200L          // 点击最大时长
        const val LONG_PRESS_TIMEOUT = 500L   // 长按触发时长
        const val SWIPE_THRESHOLD = 50f       // 滑动最小距离
        const val SWIPE_VELOCITY = 100f       // 滑动最小速度
    }
    
    // 识别手势类型
    fun recognizeGesture(
        downTime: Long,
        upTime: Long,
        downPos: Offset,
        upPos: Offset,
        velocity: Velocity
    ): GestureType {
        val duration = upTime - downTime
        val distance = (upPos - downPos).getDistance()
        
        return when {
            // 点击：时间短，距离小
            duration < TAP_TIMEOUT && distance < SWIPE_THRESHOLD -> {
                GestureType.TAP
            }
            
            // 长按：时间长，距离小
            duration >= LONG_PRESS_TIMEOUT && distance < SWIPE_THRESHOLD -> {
                GestureType.LONG_PRESS
            }
            
            // 上滑：向上滑动
            upPos.y < downPos.y - SWIPE_THRESHOLD && 
            velocity.y < -SWIPE_VELOCITY -> {
                GestureType.SWIPE_UP
            }
            
            // 下滑：向下滑动
            upPos.y > downPos.y + SWIPE_THRESHOLD && 
            velocity.y > SWIPE_VELOCITY -> {
                GestureType.SWIPE_DOWN
            }
            
            // 左滑：向左滑动
            upPos.x < downPos.x - SWIPE_THRESHOLD && 
            velocity.x < -SWIPE_VELOCITY -> {
                GestureType.SWIPE_LEFT
            }
            
            // 右滑：向右滑动
            upPos.x > downPos.x + SWIPE_THRESHOLD && 
            velocity.x > SWIPE_VELOCITY -> {
                GestureType.SWIPE_RIGHT
            }
            
            else -> GestureType.UNKNOWN
        }
    }
}

enum class GestureType {
    TAP,
    LONG_PRESS,
    SWIPE_UP,
    SWIPE_DOWN,
    SWIPE_LEFT,
    SWIPE_RIGHT,
    UNKNOWN
}
```

### 2.3 滑动输入

```kotlin
@Composable
fun SwipeInputHandler(
    onSwipeInput: (String) -> Unit
) {
    var swipePath by remember { mutableStateOf(listOf<Offset>()) }
    var isSwiping by remember { mutableStateOf(false) }
    
    Box(
        modifier = Modifier
            .fillMaxSize()
            .pointerInput(Unit) {
                detectDragGestures(
                    onDragStart = { offset ->
                        isSwiping = true
                        swipePath = listOf(offset)
                    },
                    onDrag = { change, _ ->
                        change.consume()
                        swipePath = swipePath + change.position
                    },
                    onDragEnd = {
                        isSwiping = false
                        val word = recognizeSwipePath(swipePath)
                        if (word.isNotEmpty()) {
                            onSwipeInput(word)
                        }
                        swipePath = emptyList()
                    }
                )
            }
    ) {
        // 绘制滑动轨迹
        if (isSwiping && swipePath.size >= 2) {
            Canvas(modifier = Modifier.fillMaxSize()) {
                val path = Path().apply {
                    moveTo(swipePath[0].x, swipePath[0].y)
                    for (i in 1 until swipePath.size) {
                        lineTo(swipePath[i].x, swipePath[i].y)
                    }
                }
                drawPath(
                    path = path,
                    color = Color(0xFF00D4FF).copy(alpha = 0.6f),
                    style = Stroke(
                        width = 8.dp.toPx(),
                        cap = StrokeCap.Round,
                        join = StrokeJoin.Round
                    )
                )
            }
        }
    }
}

// 识别滑动路径对应的词语
fun recognizeSwipePath(path: List<Offset>): String {
    // 遍历路径，识别经过的按键
    val traversedKeys = mutableListOf<Char>()
    var lastKey: Char? = null
    
    path.forEach { point ->
        val key = getKeyAtPoint(point)
        if (key != null && key != lastKey) {
            traversedKeys.add(key)
            lastKey = key
        }
    }
    
    // 使用词典匹配词语
    return dictionary.search(traversedKeys.joinToString(""))
}
```

### 3.3 按键反馈

```kotlin
@Composable
fun KeyFeedback(
    state: KeyInteractionState,
    hapticEnabled: Boolean = true,
    soundEnabled: Boolean = true
) {
    val hapticFeedback = LocalHapticFeedback.current
    val context = LocalContext.current
    
    LaunchedEffect(state) {
        when (state) {
            KeyInteractionState.PRESSED -> {
                // 触觉反馈
                if (hapticEnabled) {
                    hapticFeedback.performHapticFeedback(
                        HapticFeedbackType.TextHandleMove
                    )
                }
                
                // 声音反馈
                if (soundEnabled) {
                    playKeySound(context)
                }
            }
            
            KeyInteractionState.LONG_PRESSED -> {
                // 长按触觉反馈（更强）
                if (hapticEnabled) {
                    hapticFeedback.performHapticFeedback(
                        HapticFeedbackType.LongPress
                    )
                }
            }
            
            else -> { /* 无反馈 */ }
        }
    }
}

// 播放按键音
fun playKeySound(context: Context) {
    val audioManager = context.getSystemService(Context.AUDIO_SERVICE) as AudioManager
    audioManager.playSoundEffect(AudioManager.FX_KEYPRESS_STANDARD, -1f)
}
```

```kotlin
@Composable
fun SwipeableCandidate(
    candidate: String,
    onSwipeLeft: () -> Unit,  // 左滑删除
    onSwipeRight: () -> Unit  // 右滑置顶
) {
    var offsetX by remember { mutableFloatStateOf(0f) }
    var isSwiping by remember { mutableStateOf(false) }
    
    Box(
        modifier = Modifier
            .offset { IntOffset(offsetX.roundToInt(), 0) }
            .pointerInput(Unit) {
                detectHorizontalDragGestures(
                    onDragStart = {
                        isSwiping = true
                    },
                    onDragEnd = {
                        if (offsetX < -100) {
                            onSwipeLeft()
                        } else if (offsetX > 100) {
                            onSwipeRight()
                        }
                        offsetX = 0f
                        isSwiping = false
                    },
                    onHorizontalDrag = { change, dragAmount ->
                        change.consume()
                        offsetX = (offsetX + dragAmount).coerceIn(-200f, 200f)
                    }
                )
            }
    ) {
        // 滑动时显示操作图标
        if (isSwiping) {
            if (offsetX < -50) {
                // 左滑：显示删除图标
                Icon(
                    imageVector = Icons.Default.Delete,
                    contentDescription = "删除",
                    tint = Color(0xFFEF4444),
                    modifier = Modifier
                        .align(Alignment.CenterEnd)
                        .padding(end = 16.dp)
                )
            } else if (offsetX > 50) {
                // 右滑：显示置顶图标
                Icon(
                    imageVector = Icons.Default.Star,
                    contentDescription = "置顶",
                    tint = Color(0xFFF59E0B),
                    modifier = Modifier
                        .align(Alignment.CenterStart)
                        .padding(start = 16.dp)
                )
            }
        }
        
        // 候选词内容
        CandidateItem(candidate = candidate)
    }
}
```

```kotlin
@Composable
fun SpaceBarVoiceHandler(
    onVoiceStart: () -> Unit,
    onVoiceStop: () -> Unit,
    onVoiceCancel: () -> Unit
) {
    var isLongPressing by remember { mutableStateOf(false) }
    var longPressStartTime by remember { mutableLongStateOf(0L) }
    
    Box(
        modifier = Modifier

                    onDragStart = {
                        isLongPressing = true
                        longPressStartTime = System.currentTimeMillis()
                        onVoiceStart()
                    },
                    onDrag = { change, _ ->
                        change.consume()
                        // 检查是否上滑取消
                        if (change.position.y < -100) {
                            onVoiceCancel()
                            isLongPressing = false
                        }
                    },
                    onDragEnd = {
                        if (isLongPressing) {
                            onVoiceStop()
                            isLongPressing = false
                        }
                    }
                )
            }
    ) {
        SpaceKey(
            isVoiceActive = isLongPressing
        )
    }
}
```

```kotlin
        }

        ) {
```
