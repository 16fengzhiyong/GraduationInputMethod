# 布局系统设计

## 1. 设计理念

### 1.1 核心目标
打造**灵活可定制**的键盘布局系统，支持多种预设布局和完全自定义编辑，满足不同用户的使用习惯和场景需求。

### 1.2 设计原则
- **灵活性**：支持多种布局模式和自定义
- **一致性**：保持统一的视觉和交互规范
- **响应式**：适配不同屏幕尺寸
- **高效性**：快速切换和编辑

### 4.3 拖拽编辑

```kotlin
@Composable
fun DraggableKey(
    key: KeyLayout,
    isSelected: Boolean,
    onDragStart: () -> Unit,
    onDrag: (Offset) -> Unit,
    onDragEnd: () -> Unit,
    onClick: () -> Unit
) {
    var offset by remember { mutableStateOf(Offset.Zero) }
    
    Box(
        modifier = Modifier
            .offset { IntOffset(offset.x.roundToInt(), offset.y.roundToInt()) }
            .pointerInput(Unit) {
                detectDragGestures(
                    onDragStart = { onDragStart() },
                    onDrag = { change, dragAmount ->
                        change.consume()
                        offset += dragAmount
                        onDrag(dragAmount)
                    },
                    onDragEnd = {
                        offset = Offset.Zero
                        onDragEnd()
                    }
                )
            }
            .clickable { onClick() }
    ) {
        KeyButton(
            label = key.label,
            isSelected = isSelected
        )
    }
}
```

### 4.4 缩放手势

```kotlin
@Composable
fun ScalableKeyboard(
    layout: KeyboardLayout,
    onScaleChange: (Float) -> Unit
) {
    var scale by remember { mutableStateOf(1f) }
    var offset by remember { mutableStateOf(Offset.Zero) }
    
    Box(
        modifier = Modifier
            .pointerInput(Unit) {
                detectTransformGestures { _, pan, zoom, _ ->
                    scale = (scale * zoom).coerceIn(0.5f, 2f)
                    offset += pan
                    onScaleChange(scale)
                }
            }
    ) {
        Box(
            modifier = Modifier
                .graphicsLayer {
                    scaleX = scale
                    scaleY = scale
                    translationX = offset.x
                    translationY = offset.y
                }
        ) {
            KeyboardLayout(layout)
        }
    }
}
```

```kotlin
fun ResponsiveKeyboard(
    layout: KeyboardLayout,
    windowInfo: WindowInfo

    val config = when {
        // 小屏幕手机
        windowInfo.screenWidth < 360.dp -> layout.config.copy(
            keyHeight = 40.dp,
            keySpacing = 4.dp,
            rowSpacing = 4.dp,
            cornerRadius = 12.dp,
            scale = 0.85f
        )
        
        // 普通手机
        windowInfo.screenWidth < 480.dp -> layout.config.copy(
            keyHeight = 48.dp,
            keySpacing = 5.dp,
            rowSpacing = 5.dp,
            cornerRadius = 16.dp,
            scale = 1f
        )
        
        // 大屏手机/平板
        windowInfo.screenWidth < 768.dp -> layout.config.copy(
            keyHeight = 52.dp,
            keySpacing = 6.dp,
            rowSpacing = 6.dp,
            cornerRadius = 18.dp,
            scale = 1.1f
        )
        
        // 平板
        else -> layout.config.copy(
            keyHeight = 56.dp,
            keySpacing = 8.dp,
            rowSpacing = 8.dp,
            cornerRadius = 20.dp,
            scale = 1.2f
        )
    }
    
    KeyboardLayout(
        layout = layout.copy(config = config)
    )
}
```

### 7.2 切换动画实现

```kotlin
@Composable
fun AnimatedLayoutSwitch(
    currentLayout: KeyboardLayout,
    targetLayout: KeyboardLayout,
    transitionType: LayoutTransitionType,
    onTransitionComplete: () -> Unit
) {
    var transitionState by remember { mutableStateOf(0f) }
    
    LaunchedEffect(targetLayout) {
        animate(
            initialValue = 0f,
            targetValue = 1f,
            animationSpec = tween(
                durationMillis = 300,
                easing = FastOutSlowInEasing
            )
        ) { value ->
            transitionState = value
        }
        onTransitionComplete()
    }
    
    Box(modifier = Modifier.fillMaxSize()) {
        // 当前布局
        Box(
            modifier = Modifier
                .graphicsLayer {
                    when (transitionType) {
                        LayoutTransitionType.SLIDE_LEFT -> {
                            translationX = -size.width * transitionState
                            alpha = 1f - transitionState
                        }
                        LayoutTransitionType.FADE -> {
                            alpha = 1f - transitionState
                        }
                        LayoutTransitionType.SCALE -> {
                            scaleX = 1f - transitionState * 0.5f
                            scaleY = 1f - transitionState * 0.5f
                            alpha = 1f - transitionState
                        }
                        // ... 其他动画类型
                    }
                }
        ) {
            KeyboardLayout(layout = currentLayout)
        }
        
        // 目标布局
        Box(
            modifier = Modifier
                .graphicsLayer {
                    when (transitionType) {
                        LayoutTransitionType.SLIDE_LEFT -> {
                            translationX = size.width * (1f - transitionState)
                            alpha = transitionState
                        }
                        LayoutTransitionType.FADE -> {
                            alpha = transitionState
                        }
                        LayoutTransitionType.SCALE -> {
                            scaleX = 0.5f + transitionState * 0.5f
                            scaleY = 0.5f + transitionState * 0.5f
                            alpha = transitionState
                        }
                        // ... 其他动画类型
                    }
                }
        ) {
            KeyboardLayout(layout = targetLayout)
        }
    }
}
```

```kotlin
class LayoutManager {
    // 导出布局
    fun exportLayout(layout: KeyboardLayout): String {
        val json = Json { 
            prettyPrint = true
            ignoreUnknownKeys = true
        }
        return json.encodeToString(layout)
    }
    
    // 导入布局
    fun importLayout(json: String): KeyboardLayout {
        return Json.decodeFromString(json)
    }
    
    // 保存到文件
    fun saveToFile(layout: KeyboardLayout, file: File) {
        file.writeText(exportLayout(layout))
    }
    
    // 从文件加载
    fun loadFromFile(file: File): KeyboardLayout {
        return importLayout(file.readText())
    }
    
    // 分享布局
    fun shareLayout(context: Context, layout: KeyboardLayout) {
        val json = exportLayout(layout)
        val intent = Intent(Intent.ACTION_SEND).apply {
            type = "application/json"
            putExtra(Intent.EXTRA_TEXT, json)
        }
        context.startActivity(intent)
    }
}
```

```kotlin
class LayoutSyncManager(

    }

        val json = apiService.downloadLayout(layoutId)
        return layoutManager.importLayout(json)
    }
    
    // 同步所有布局
    suspend fun syncAllLayouts() {
        val localLayouts = layoutDao.getAllLayouts().first()
        val remoteLayouts = apiService.getLayouts()
        
        // 合并本地和远程布局
        val mergedLayouts = mergeLayouts(localLayouts, remoteLayouts)
        
        // 保存到本地
        mergedLayouts.forEach { layout ->
            layoutDao.insertLayout(layout)
        }
    }
}
```
