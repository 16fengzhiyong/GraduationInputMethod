# 主题系统设计

## 1. 设计理念

### 1.1 核心目标
构建**灵活、可扩展、易定制**的主题系统，让用户能够轻松打造个性化的输入法外观，同时保持视觉一致性和性能优化。

### 1.2 设计原则
- **模块化**：主题组件独立，可自由组合
- **可扩展**：支持自定义主题和在线主题商店
- **性能优化**：主题切换无卡顿
- **向后兼容**：新主题不影响旧版本

```kotlin
    val keyBackgroundPressed: Brush,

    val keyBackgroundDisabled: Color,

    val textPrimary: Color,

    val accentLight: Color,
    val accentDark: Color,
    
    // 语义色
    val success: Color,
    val warning: Color,
    val error: Color,
    val info: Color,
    
    // 特效色

    val shadow: Color,
    val highlight: Color
)

    val primaryFont: FontFamily,
    val secondaryFont: FontFamily,
    val keyLabelSize: TextUnit,
    val candidateSize: TextUnit,

    val fontWeight: FontWeight
)

    val keyHeight: Dp,
    val keySpacing: Dp,
```

## 4. 主题管理器

### 4.1 主题管理器实现

```kotlin
class ThemeManager(
    private val context: Context,
    private val dataStore: DataStore<Preferences>
) {
    // 当前主题
    private val _currentTheme = MutableStateFlow(DeepSpaceTheme)
    val currentTheme: StateFlow<KeyboardTheme> = _currentTheme.asStateFlow()
    
    // 可用主题列表
    private val _availableThemes = MutableStateFlow<List<KeyboardTheme>>(emptyList())
    val availableThemes: StateFlow<List<KeyboardTheme>> = _availableThemes.asStateFlow()
    
    // 主题缓存
    private val themeCache = mutableMapOf<String, KeyboardTheme>()
    
    init {
        // 加载内置主题
        loadBuiltInThemes()
        
        // 加载自定义主题
        loadCustomThemes()
        
        // 恢复上次使用的主题
        restoreLastTheme()
    }
    
    // 切换主题
    suspend fun switchTheme(themeId: String) {
        val theme = themeCache[themeId] ?: return
        
        // 保存当前主题ID
        dataStore.edit { preferences ->
            preferences[THEME_ID_KEY] = themeId
        }
        
        // 更新当前主题
        _currentTheme.value = theme
    }
    
    // 加载内置主题
    private fun loadBuiltInThemes() {
        val builtInThemes = listOf(
            DeepSpaceTheme,
            NeonTheme,
            CyberTheme,
            AuroraTheme
        )
        
        builtInThemes.forEach { theme ->
            themeCache[theme.id] = theme
        }
        
        _availableThemes.value = builtInThemes
    }
    
    // 加载自定义主题
    private fun loadCustomThemes() {
        // 从本地存储加载自定义主题
        val customThemesDir = File(context.filesDir, "themes")
        if (customThemesDir.exists()) {
            customThemesDir.listFiles()?.forEach { file ->
                try {
                    val json = file.readText()
                    val theme = Json.decodeFromString<KeyboardTheme>(json)
                    themeCache[theme.id] = theme
                } catch (e: Exception) {
                    // 忽略损坏的主题文件
                }
            }
        }
        
        _availableThemes.value = themeCache.values.toList()
    }
    
    // 恢复上次使用的主题
    private fun restoreLastTheme() {
        CoroutineScope(Dispatchers.IO).launch {
            dataStore.data.first()[THEME_ID_KEY]?.let { themeId ->
                themeCache[themeId]?.let { theme ->
                    _currentTheme.value = theme
                }
            }
        }
    }
    
    // 导入主题
    suspend fun importTheme(json: String): Result<KeyboardTheme> {
        return try {
            val theme = Json.decodeFromString<KeyboardTheme>(json)
            
            // 验证主题
            if (!validateTheme(theme)) {
                return Result.failure(Exception("主题格式无效"))
            }
            
            // 保存主题
            saveTheme(theme)
            
            // 添加到缓存
            themeCache[theme.id] = theme
            _availableThemes.value = themeCache.values.toList()
            
            Result.success(theme)
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
    
    // 导出主题
    fun exportTheme(themeId: String): String? {
        val theme = themeCache[themeId] ?: return null
        return Json.encodeToString(theme)
    }
    
    // 保存主题
    private suspend fun saveTheme(theme: KeyboardTheme) {
        val customThemesDir = File(context.filesDir, "themes")
        if (!customThemesDir.exists()) {
            customThemesDir.mkdirs()
        }
        
        val themeFile = File(customThemesDir, "${theme.id}.json")
        themeFile.writeText(Json.encodeToString(theme))
    }
    
    // 验证主题
    private fun validateTheme(theme: KeyboardTheme): Boolean {
        return theme.id.isNotEmpty() &&
               theme.name.isNotEmpty() &&
               theme.colors != null &&
               theme.typography != null &&
               theme.layout != null
    }
    
    // 删除主题
    suspend fun deleteTheme(themeId: String) {
        // 不能删除内置主题
        if (isBuiltInTheme(themeId)) return
        
        // 从缓存移除
        themeCache.remove(themeId)
        _availableThemes.value = themeCache.values.toList()
        
        // 删除文件
        val themeFile = File(context.filesDir, "themes/$themeId.json")
        if (themeFile.exists()) {
            themeFile.delete()
        }
        
        // 如果删除的是当前主题，切换到默认主题
        if (_currentTheme.value.id == themeId) {
            switchTheme(DeepSpaceTheme.id)
        }
    }
    
    // 检查是否为内置主题
    private fun isBuiltInTheme(themeId: String): Boolean {
        return themeId in listOf(
            DeepSpaceTheme.id,
            NeonTheme.id,
            CyberTheme.id,
            AuroraTheme.id
        )
    }
    
    companion object {
        private val THEME_ID_KEY = stringPreferencesKey("theme_id")
    }
}
```

## 5. 主题切换

### 5.1 主题切换动画

```kotlin
@Composable
fun AnimatedThemeSwitch(
    targetTheme: KeyboardTheme,
    content: @Composable () -> Unit
) {
    var currentTheme by remember { mutableStateOf(targetTheme) }
    var transitionProgress by remember { mutableFloatStateOf(0f) }
    
    LaunchedEffect(targetTheme) {
        if (currentTheme.id != targetTheme.id) {
            // 开始过渡
            animate(
                initialValue = 0f,
                targetValue = 1f,
                animationSpec = tween(300)
            ) { value ->
                transitionProgress = value
            }
            
            // 完成过渡
            currentTheme = targetTheme
            transitionProgress = 0f
        }
    }
    
    // 创建过渡色彩
    val transitionColors = if (transitionProgress > 0f) {
        interpolateColors(currentTheme.colors, targetTheme.colors, transitionProgress)
    } else {
        currentTheme.colors
    }
    
    CompositionLocalProvider(
        LocalKeyboardTheme provides currentTheme.copy(colors = transitionColors)
    ) {
        content()
    }
}

// 色彩插值
fun interpolateColors(
    from: ThemeColors,
    to: ThemeColors,
    progress: Float
): ThemeColors {
    return ThemeColors(
        background = interpolateBrush(from.background, to.background, progress),
        backgroundSecondary = lerp(from.backgroundSecondary, to.backgroundSecondary, progress),
        backgroundTertiary = lerp(from.backgroundTertiary, to.backgroundTertiary, progress),
        // ... 其他色彩插值
    )
}
```

```kotlin
class ThemeImportExportManager(
    private val context: Context,
    private val themeManager: ThemeManager
) {
    // 导出主题
    suspend fun exportTheme(themeId: String): Uri? {
        val theme = themeManager.currentTheme.value
        val json = Json.encodeToString(theme)
        
        // 创建临时文件
        val file = File(context.cacheDir, "theme_${theme.id}.json")
        file.writeText(json)
        
        // 返回文件URI
        return FileProvider.getUriForFile(
            context,
            "${context.packageName}.fileprovider",
            file
        )
    }
    
    // 导入主题
    suspend fun importTheme(uri: Uri): Result<KeyboardTheme> {
        return try {
            // 读取文件内容
            val json = context.contentResolver.openInputStream(uri)?.use { inputStream ->
                inputStream.bufferedReader().readText()
            } ?: return Result.failure(Exception("无法读取文件"))
            
            // 导入主题
            themeManager.importTheme(json)
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
    
    // 分享主题
    fun shareTheme(themeId: String) {
        CoroutineScope(Dispatchers.IO).launch {
            val uri = exportTheme(themeId) ?: return@launch
            
            val intent = Intent(Intent.ACTION_SEND).apply {
                type = "application/json"
                putExtra(Intent.EXTRA_STREAM, uri)
                addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
            }
            
            context.startActivity(
                Intent.createChooser(intent, "分享主题")
            )
        }
    }
    
    // 从剪贴板导入
    suspend fun importFromClipboard(): Result<KeyboardTheme> {
        val clipboardManager = context.getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager
        val clipData = clipboardManager.primaryClip ?: return Result.failure(Exception("剪贴板为空"))
        
        val text = clipData.getItemAt(0).text?.toString()
            ?: return Result.failure(Exception("剪贴板中没有文本"))
        
        return themeManager.importTheme(text)
    }
}
```
