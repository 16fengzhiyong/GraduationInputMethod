# 组件规范文档

## 1. 设计理念

### 1.1 核心目标
建立一套**统一、可复用、可扩展**的组件系统，确保整个输入法界面的视觉一致性和开发效率。

### 1.2 设计原则
- **一致性**：统一的视觉语言和交互模式
- **可复用性**：组件可在不同场景复用
- **可定制性**：支持主题和配置定制
- **性能优化**：组件轻量高效

```kotlin
@Composable
fun OmeletteButton(
    text: String,
    onClick: () -> Unit,
    modifier: Modifier = Modifier,
    variant: ButtonVariant = ButtonVariant.PRIMARY,
    size: ButtonSize = ButtonSize.MEDIUM,
    enabled: Boolean = true,
    icon: ImageVector? = null
) {
    val theme = keyboardTheme()
    
    val colors = when (variant) {
        ButtonVariant.PRIMARY -> ButtonColors(
            background = Brush.horizontalGradient(
                colors = listOf(theme.colors.accent, theme.colors.accentDark)
            ),
            contentColor = Color.White,
            disabledBackground = theme.colors.textDisabled,
            disabledContentColor = theme.colors.textDisabled
        )
        ButtonVariant.SECONDARY -> ButtonColors(
            background = Brush.horizontalGradient(
                colors = listOf(
                    theme.colors.backgroundSecondary,
                    theme.colors.backgroundTertiary
                )
            ),
            contentColor = theme.colors.textPrimary,
            disabledBackground = theme.colors.textDisabled,
            disabledContentColor = theme.colors.textDisabled
        )
        ButtonVariant.OUTLINED -> ButtonColors(
            background = Brush.linearGradient(
                colors = listOf(Color.Transparent, Color.Transparent)
            ),
            contentColor = theme.colors.accent,
            disabledBackground = Color.Transparent,
            disabledContentColor = theme.colors.textDisabled
        )
        ButtonVariant.GHOST -> ButtonColors(
            background = Brush.linearGradient(
                colors = listOf(Color.Transparent, Color.Transparent)
            ),
            contentColor = theme.colors.textPrimary,
            disabledBackground = Color.Transparent,
            disabledContentColor = theme.colors.textDisabled
        )
    }
    
    val height = when (size) {
        ButtonSize.SMALL -> 32.dp
        ButtonSize.MEDIUM -> 40.dp
        ButtonSize.LARGE -> 48.dp
    }
    
    val fontSize = when (size) {
        ButtonSize.SMALL -> 12.sp
        ButtonSize.MEDIUM -> 14.sp
        ButtonSize.LARGE -> 16.sp
    }
    
    val padding = when (size) {
        ButtonSize.SMALL -> PaddingValues(horizontal = 12.dp, vertical = 4.dp)
        ButtonSize.MEDIUM -> PaddingValues(horizontal = 16.dp, vertical = 8.dp)
        ButtonSize.LARGE -> PaddingValues(horizontal = 24.dp, vertical = 12.dp)
    }
    
    var isPressed by remember { mutableStateOf(false) }
    
    Box(
        modifier = modifier
            .height(height)
            .clip(RoundedCornerShape(height / 2))
            .background(
                brush = if (enabled) colors.background else {
                    Brush.linearGradient(
                        colors = listOf(
                            colors.disabledBackground,
                            colors.disabledBackground
                        )
                    )
                }
            )
            .then(
                if (variant == ButtonVariant.OUTLINED) {
                    Modifier.border(
                        width = 1.dp,
                        color = if (enabled) theme.colors.accent else theme.colors.textDisabled,
                        shape = RoundedCornerShape(height / 2)
                    )
                } else {
                    Modifier
                }
            )
            .clickable(enabled = enabled) {
                isPressed = true
                onClick()
            }
            .padding(padding),
        contentAlignment = Alignment.Center
    ) {
        Row(
            horizontalArrangement = Arrangement.Center,
            verticalAlignment = Alignment.CenterVertically
        ) {
            if (icon != null) {
                Icon(
                    imageVector = icon,
                    contentDescription = null,
                    tint = if (enabled) colors.contentColor else colors.disabledContentColor,
                    modifier = Modifier.size(fontSize.value.dp)
                )
                Spacer(modifier = Modifier.width(8.dp))
            }
            
            Text(
                text = text,
                color = if (enabled) colors.contentColor else colors.disabledContentColor,
                fontSize = fontSize,
                fontWeight = FontWeight.Medium
            )
        }
    }
}

    SECONDARY,
    OUTLINED,
    GHOST
}

    MEDIUM,
    LARGE
```

```kotlin
@Composable
fun OmeletteTextField(
    value: String,
    onValueChange: (String) -> Unit,
    modifier: Modifier = Modifier,
    label: String? = null,
    placeholder: String? = null,
    leadingIcon: ImageVector? = null,
    trailingIcon: ImageVector? = null,
    onTrailingIconClick: (() -> Unit)? = null,
    isError: Boolean = false,
    errorMessage: String? = null,
    enabled: Boolean = true
) {
    val theme = keyboardTheme()
    
    var isFocused by remember { mutableStateOf(false) }
    
    Column(modifier = modifier) {
        if (label != null) {
            Text(
                text = label,
                style = MaterialTheme.typography.bodySmall,
                color = if (isError) {
                    theme.colors.error
                } else if (isFocused) {
                    theme.colors.accent
                } else {
                    theme.colors.textSecondary
                },
                modifier = Modifier.padding(bottom = 4.dp)
            )
        }
        
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .height(48.dp)
                .clip(RoundedCornerShape(12.dp))
                .background(
                    color = theme.colors.backgroundSecondary
                )
                .border(
                    width = if (isFocused) 2.dp else 1.dp,
                    color = when {
                        isError -> theme.colors.error
                        isFocused -> theme.colors.accent
                        else -> theme.colors.textTertiary.copy(alpha = 0.3f)
                    },
                    shape = RoundedCornerShape(12.dp)
                )
        ) {
            Row(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(horizontal = 12.dp),
                verticalAlignment = Alignment.CenterVertically
            ) {
                if (leadingIcon != null) {
                    Icon(
                        imageVector = leadingIcon,
                        contentDescription = null,
                        tint = theme.colors.textTertiary,
                        modifier = Modifier.size(20.dp)
                    )
                    Spacer(modifier = Modifier.width(8.dp))
                }
                
                BasicTextField(
                    value = value,
                    onValueChange = onValueChange,
                    enabled = enabled,
                    textStyle = MaterialTheme.typography.bodyLarge.copy(
                        color = if (enabled) {
                            theme.colors.textPrimary
                        } else {
                            theme.colors.textDisabled
                        }
                    ),
                    modifier = Modifier
                        .weight(1f)
                        .onFocusChanged { focusState ->
                            isFocused = focusState.isFocused
                        },
                    decorationBox = { innerTextField ->
                        Box {
                            if (value.isEmpty() && placeholder != null) {
                                Text(
                                    text = placeholder,
                                    style = MaterialTheme.typography.bodyLarge,
                                    color = theme.colors.textTertiary
                                )
                            }
                            innerTextField()
                        }
                    }
                )
                
                if (trailingIcon != null) {
                    Spacer(modifier = Modifier.width(8.dp))
                    IconButton(
                        onClick = { onTrailingIconClick?.invoke() },
                        modifier = Modifier.size(20.dp)
                    ) {
                        Icon(
                            imageVector = trailingIcon,
                            contentDescription = null,
                            tint = theme.colors.textTertiary
                        )
                    }
                }
            }
        }
        
        if (isError && errorMessage != null) {
            Text(
                text = errorMessage,
                style = MaterialTheme.typography.bodySmall,
                color = theme.colors.error,
                modifier = Modifier.padding(top = 4.dp, start = 12.dp)
            )
        }
    }
}
```

```kotlin
@Composable
fun OmeletteCard(
    modifier: Modifier = Modifier,
    variant: CardVariant = CardVariant.DEFAULT,
    onClick: (() -> Unit)? = null,
    content: @Composable () -> Unit
) {
    val theme = keyboardTheme()
    
    val backgroundColor = when (variant) {
        CardVariant.DEFAULT -> theme.colors.backgroundSecondary
        CardVariant.ELEVATED -> theme.colors.backgroundSecondary
        CardVariant.OUTLINED -> Color.Transparent
        CardVariant.FILLED -> theme.colors.backgroundTertiary
    }
    
    val elevation = when (variant) {
        CardVariant.DEFAULT -> 0.dp
        CardVariant.ELEVATED -> 4.dp
        CardVariant.OUTLINED -> 0.dp
        CardVariant.FILLED -> 0.dp
    }
    
    val border = when (variant) {
        CardVariant.OUTLINED -> Modifier.border(
            width = 1.dp,
            color = theme.colors.textTertiary.copy(alpha = 0.3f),
            shape = RoundedCornerShape(16.dp)
        )
        else -> Modifier
    }
    
    Box(
        modifier = modifier
            .fillMaxWidth()
            .shadow(
                elevation = elevation,
                shape = RoundedCornerShape(16.dp),
                ambientColor = Color.Black.copy(alpha = 0.1f),
                spotColor = Color.Black.copy(alpha = 0.1f)
            )
            .then(border)
            .clip(RoundedCornerShape(16.dp))
            .background(
                color = backgroundColor,
                shape = RoundedCornerShape(16.dp)
            )
            .then(
                if (onClick != null) {
                    Modifier.clickable { onClick() }
                } else {
                    Modifier
                }
            )
    ) {
        content()
    }
}

enum class CardVariant {
    DEFAULT,
    ELEVATED,
    OUTLINED,
    FILLED
}
```

```kotlin
@Composable
fun KeyButton(
    key: KeyLayout,
    modifier: Modifier = Modifier,
    state: KeyState = KeyState.IDLE,
    theme: KeyboardTheme = keyboardTheme(),
    onKeyPress: ((KeyAction) -> Unit)? = null,
    onKeyLongPress: ((KeyAction) -> Unit)? = null,
    onKeySwipeUp: ((KeyAction) -> Unit)? = null
) {
    val colors = getKeyColors(state, theme)
    val shape = RoundedCornerShape(theme.layout.cornerRadius)
    
    // 按压缩放动画
    val scale by animateFloatAsState(
        targetValue = if (state == KeyState.PRESSED) 0.95f else 1f,
        animationSpec = spring(
            dampingRatio = Spring.DampingRatioMediumBouncy,
            stiffness = Spring.StiffnessLow
        )
    )
    
    Box(
        modifier = modifier
            .size(
                width = getKeyWidth(key.width, theme),
                height = key.height ?: theme.layout.keyHeight
            )
            .graphicsLayer {
                scaleX = scale
                scaleY = scale
            }
            .keyShadow(
                elevation = if (state == KeyState.PRESSED) 1.dp else 4.dp,
                isPressed = state == KeyState.PRESSED,
                theme = theme
            )
            .background(
                brush = colors.background,
                shape = shape
            )
            .keyHighlight(
                alpha = if (state == KeyState.PRESSED) 0.05f else 0.15f,
                theme = theme
            )
            .neonBorder(
                color = theme.colors.accent,
                alpha = colors.borderAlpha,
                shape = shape
            )
            .pointerInput(Unit) {
                detectTapGestures(
                    onPress = {
                        onKeyPress?.invoke(key.action)
                    },
                    onLongPress = {
                        onKeyLongPress?.invoke(key.action)
                    }
                )
            }
            .pointerInput(Unit) {
                detectVerticalDragGestures(
                    onDragEnd = { /* ... */ },
                    onVerticalDrag = { change, dragAmount ->
                        if (dragAmount < -50) {
                            onKeySwipeUp?.invoke(key.action)
                        }
                    }
                )
            },
        contentAlignment = Alignment.Center
    ) {
        // 按键标签
        Text(
            text = key.label,
            color = colors.textColor,
            fontSize = theme.typography.keyLabelSize,
            fontWeight = theme.typography.fontWeight,
            textAlign = TextAlign.Center
        )
        
        // 副标签（九宫格）
        if (key.subLabel != null) {
            Text(
                text = key.subLabel!!,
                color = colors.textColor.copy(alpha = 0.6f),
                fontSize = theme.typography.keyLabelSize * 0.5f,
                modifier = Modifier
                    .align(Alignment.BottomCenter)
                    .padding(bottom = 4.dp)
            )
        }
        
        // 上滑符号
        if (key.swipeUpSymbol != null) {
            Text(
                text = key.swipeUpSymbol!!,
                color = colors.textColor.copy(alpha = 0.4f),
                fontSize = 8.sp,
                modifier = Modifier
                    .align(Alignment.TopCenter)
                    .padding(top = 2.dp)
            )
        }
    }
}

// 获取按键颜色
@Composable
fun getKeyColors(state: KeyState, theme: KeyboardTheme): KeyColors {
    return when (state) {
        KeyState.IDLE -> KeyColors(
            background = theme.colors.keyBackground,
            textColor = theme.colors.textPrimary,
            shadowAlpha = 0.3f,
            highlightAlpha = 0.15f,
            borderAlpha = 0.3f
        )
        KeyState.HOVERED -> KeyColors(
            background = theme.colors.keyBackgroundHover,
            textColor = theme.colors.textPrimary,
            shadowAlpha = 0.4f,
            highlightAlpha = 0.2f,
            borderAlpha = 0.5f
        )
        KeyState.PRESSED -> KeyColors(
            background = theme.colors.keyBackgroundPressed,
            textColor = theme.colors.textPrimary.copy(alpha = 0.8f),
            shadowAlpha = 0.1f,
            highlightAlpha = 0.05f,
            borderAlpha = 0.6f
        )
        KeyState.LONG_PRESSED -> KeyColors(
            background = theme.colors.keyBackgroundPressed,
            textColor = theme.colors.accent,
            shadowAlpha = 0.2f,
            highlightAlpha = 0.1f,
            borderAlpha = 0.8f
        )
        KeyState.DISABLED -> KeyColors(
            background = Brush.linearGradient(
                colors = listOf(
                    theme.colors.keyBackgroundDisabled,
                    theme.colors.keyBackgroundDisabled
                )
            ),
            textColor = theme.colors.textDisabled,
            shadowAlpha = 0.1f,
            highlightAlpha = 0.05f,
            borderAlpha = 0.1f
        )
    }
}

data class KeyColors(
    val background: Brush,
    val textColor: Color,
    val shadowAlpha: Float,
    val highlightAlpha: Float,
    val borderAlpha: Float
)
```

```kotlin
@Composable
fun CandidateItem(

    isSelected: Boolean,
    onClick: () -> Unit,

                },
                fontSize = theme.typography.candidateSize,

                    FontWeight.Bold
                } else {
                    FontWeight.Normal
                }

                Text(
                    text = candidate.pinyin!!,

                    fontSize = 10.sp
                )
            }
        }
    }
```

```kotlin
@Composable
fun RippleEffect(
    color: Color,
    center: Offset,
    maxRadius: Float,
    onComplete: () -> Unit
) {
    var radius by remember { mutableFloatStateOf(0f) }
    var alpha by remember { mutableFloatStateOf(0.3f) }
    
    LaunchedEffect(Unit) {
        animate(
            initialValue = 0f,
            targetValue = maxRadius,
            animationSpec = tween(300)
        ) { value ->
            radius = value
            alpha = 0.3f * (1 - value / maxRadius)
        }
        onComplete()
    }
    
    Canvas(modifier = Modifier.fillMaxSize()) {
        drawCircle(
            color = color.copy(alpha = alpha),
            radius = radius,
            center = center
        )
    }
}
```

```kotlin
@Composable
fun PulseAnimation(
    color: Color,
    size: Dp = 48.dp,
    pulseCount: Int = 3
) {
    val infiniteTransition = rememberInfiniteTransition()
    
    val pulseAnimations = List(pulseCount) { index ->
        val delay = index * 300
        
        val scale by infiniteTransition.animateFloat(
            initialValue = 1f,
            targetValue = 1.5f,
            animationSpec = infiniteRepeatable(
                animation = tween(1000, delayMillis = delay),
                repeatMode = RepeatMode.Restart
            )
        )
        
        val alpha by infiniteTransition.animateFloat(
            initialValue = 0.5f,
            targetValue = 0f,
            animationSpec = infiniteRepeatable(
                animation = tween(1000, delayMillis = delay),
                repeatMode = RepeatMode.Restart
            )
        )
        
        Pair(scale, alpha)
    }
    
    Box(
        modifier = Modifier.size(size),
        contentAlignment = Alignment.Center
    ) {
        pulseAnimations.forEach { (scale, alpha) ->
            Box(
                modifier = Modifier
                    .size(size)
                    .graphicsLayer {
                        scaleX = scale
                        scaleY = scale
                        this.alpha = alpha
                    }
                    .background(
                        color = color,
                        shape = CircleShape
                    )
            )
        }
    }
}
```

```kotlin
@Composable
fun FeedbackAnimation(
    type: FeedbackType,
    isVisible: Boolean,
    onComplete: () -> Unit
) {
    var scale by remember { mutableFloatStateOf(0f) }
    var alpha by remember { mutableFloatStateOf(0f) }
    
    LaunchedEffect(isVisible) {
        if (isVisible) {
            // 弹出
            animate(
                initialValue = 0f,
                targetValue = 1f,
                animationSpec = spring(
                    dampingRatio = Spring.DampingRatioMediumBouncy,
                    stiffness = Spring.StiffnessLow
                )
            ) { value ->
                scale = value
                alpha = value
            }
            
            // 停留
            delay(500)
            
            // 消失
            animate(
                initialValue = 1f,
                targetValue = 0f,
                animationSpec = tween(200)
            ) { value ->
                alpha = value
            }
            
            onComplete()
        }
    }
    
    if (isVisible) {
        Box(
            modifier = Modifier
                .size(48.dp)
                .graphicsLayer {
                    scaleX = scale
                    scaleY = scale
                    this.alpha = alpha
                }
                .background(
                    color = when (type) {
                        FeedbackType.SUCCESS -> Color(0xFF10B981)
```

```kotlin
@Composable
fun SettingItem(
    title: String,
    subtitle: String? = null,
    icon: ImageVector? = null,
    trailing: @Composable (() -> Unit)? = null,
    onClick: (() -> Unit)? = null
) {
    val theme = keyboardTheme()
    
    var isPressed by remember { mutableStateOf(false) }
    
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .then(
                if (onClick != null) {
                    Modifier.pointerInput(Unit) {
                        detectTapGestures(
                            onPress = {
                                isPressed = true
                                tryAwaitRelease()
                                isPressed = false
                            },
                            onTap = { onClick() }
                        )
                    }
                } else {
                    Modifier
                }
            )
            .background(
                color = if (isPressed) {
                    theme.colors.backgroundTertiary
                } else {
                    Color.Transparent
                }
            )
            .padding(horizontal = 16.dp, vertical = 12.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        if (icon != null) {
            Icon(
                imageVector = icon,
                contentDescription = null,
                tint = theme.colors.accent,
                modifier = Modifier.size(24.dp)
            )
            Spacer(modifier = Modifier.width(16.dp))
        }
        
        Column(
            modifier = Modifier.weight(1f)
        ) {
            Text(
                text = title,
                style = MaterialTheme.typography.bodyLarge,
                color = theme.colors.textPrimary
            )
            
            if (subtitle != null) {
                Text(
                    text = subtitle,
                    style = MaterialTheme.typography.bodySmall,
                    color = theme.colors.textSecondary
                )
            }
        }
        
        if (trailing != null) {
            trailing()
        }
    }
}
```
