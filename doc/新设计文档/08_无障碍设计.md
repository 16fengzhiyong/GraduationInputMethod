# 无障碍设计规范

## 1. 设计理念

### 1.1 核心目标
确保所有用户，包括视力、听力、运动和认知障碍用户，都能**平等、有效地**使用输入法。

### 1.2 设计原则
- **可感知**：信息对所有用户可感知
- **可操作**：界面对所有用户可操作
- **可理解**：内容对所有用户可理解
- **健壮性**：与辅助技术兼容

### 1.3 标准遵循
- **WCAG 2.1 AA**：Web内容无障碍指南
- **Android Accessibility**：Android无障碍最佳实践
- **BFSG**：德国无障碍强化法（可选）

```kotlin
// 对比度检查工具
object ContrastChecker {
    // WCAG AA 标准
    const val MIN_CONTRAST_NORMAL = 4.5f    // 正文文字
    const val MIN_CONTRAST_LARGE = 3f       // 大文字（18px+或14px粗体）
    const val MIN_CONTRAST_GRAPHICAL = 3f   // 图形和UI组件
    
    // WCAG AAA 标准
    const val AAA_CONTRAST_NORMAL = 7f
    const val AAA_CONTRAST_LARGE = 4.5f
    
    // 计算相对亮度
    fun calculateLuminance(color: Color): Float {
        val r = linearizeColorComponent(color.red)
        val g = linearizeColorComponent(color.green)
        val b = linearizeColorComponent(color.blue)
        return 0.2126f * r + 0.7152f * g + 0.0722f * b
    }
    
    // 线性化颜色分量
    private fun linearizeColorComponent(component: Float): Float {
        return if (component <= 0.03928f) {
            component / 12.92f
        } else {
            ((component + 0.055f) / 1.055f).pow(2.4f)
        }
    }
    
    // 计算对比度
    fun calculateContrastRatio(color1: Color, color2: Color): Float {
        val luminance1 = calculateLuminance(color1)
        val luminance2 = calculateLuminance(color2)
        val lighter = maxOf(luminance1, luminance2)
        val darker = minOf(luminance1, luminance2)
        return (lighter + 0.05f) / (darker + 0.05f)
    }
    
    // 检查是否符合标准
    fun meetsWCAG_AA(
        textColor: Color,
        backgroundColor: Color,
        isLargeText: Boolean = false
    ): Boolean {
        val contrastRatio = calculateContrastRatio(textColor, backgroundColor)
        return if (isLargeText) {
            contrastRatio >= MIN_CONTRAST_LARGE
        } else {
            contrastRatio >= MIN_CONTRAST_NORMAL
        }
    }
    
    fun meetsWCAG_AAA(
        textColor: Color,
        backgroundColor: Color,
        isLargeText: Boolean = false
    ): Boolean {
        val contrastRatio = calculateContrastRatio(textColor, backgroundColor)
        return if (isLargeText) {
            contrastRatio >= AAA_CONTRAST_LARGE
        } else {
            contrastRatio >= AAA_CONTRAST_NORMAL
        }
    }
}

            fontSize = if (isHighContrast) {
                theme.typography.keyLabelSize * 1.2f
            } else {
                theme.typography.keyLabelSize
            },
            fontWeight = if (isHighContrast) FontWeight.Bold else theme.typography.fontWeight
        )
    }
}

@Composable
fun SoundFeedback(
    isEnabled: Boolean,
    content: @Composable () -> Unit
) {
    val context = LocalContext.current
    val audioManager = remember {
        context.getSystemService(Context.AUDIO_SERVICE) as AudioManager
    }
    
    // 按键音效
    fun playKeySound() {
        if (isEnabled) {
            audioManager.playSoundEffect(
                AudioManager.FX_KEYPRESS_STANDARD,
                -1f
            )
        }
    }
    
    // 错误音效
    fun playErrorSound() {
        if (isEnabled) {
            audioManager.playSoundEffect(
                AudioManager.FX_KEYPRESS_DELETE,
                -1f
            )
        }
    }
    
    // 成功音效
    fun playSuccessSound() {
        if (isEnabled) {
            audioManager.playSoundEffect(
                AudioManager.FX_KEYPRESS_SPACEBAR,
                -1f
            )
        }
    }
    
    CompositionLocalProvider(
        LocalSoundFeedback provides SoundFeedbackProvider(
            playKeySound = ::playKeySound,
            playErrorSound = ::playErrorSound,
            playSuccessSound = ::playSuccessSound
        )
    ) {
        content()
    }
}

)

@Composable
fun AccidentalTouchPrevention(
    isEnabled: Boolean,
    content: @Composable () -> Unit
) {
    var lastTapTime by remember { mutableLongStateOf(0L) }
    var lastTapPosition by remember { mutableStateOf(Offset.Zero) }
    
    val tapDebounceMs = if (isEnabled) 300L else 0L
    val minTapDistance = if (isEnabled) 20f else 0f
    
    Box(
        modifier = Modifier
            .pointerInput(Unit) {
                detectTapGestures { offset ->
                    val currentTime = System.currentTimeMillis()
                    val timeDiff = currentTime - lastTapTime
                    val distance = (offset - lastTapPosition).getDistance()
                    
                    if (timeDiff > tapDebounceMs || distance > minTapDistance) {
                        // 有效点击
                        lastTapTime = currentTime
                        lastTapPosition = offset
                        // 处理点击
                    }
                    // 否则视为误触，忽略
                }
            }
    ) {
        content()
    }
}
```

            .onKeyEvent { keyEvent ->
                when (keyEvent.key) {

                        moveFocusUp()
                        true
                    }
                    Key.DirectionDown -> {
                        // 移动焦点到下一行
                        moveFocusDown()
                        true
                    }
                    Key.DirectionLeft -> {
                        // 移动焦点到左边
                        moveFocusLeft()
                        true
                    }
                    Key.DirectionRight -> {
                        // 移动焦点到右边
                        moveFocusRight()
                        true
                    }
                    Key.Enter -> {
                        // 选中当前焦点按键
                        focusedKey?.let { selectKey(it) }
                        true
                    }
                    else -> false
                }

@Composable
fun ScreenReaderFeedback(
    content: @Composable () -> Unit
) {
    val accessibilityManager = LocalContext.current.getSystemService(
        Context.ACCESSIBILITY_SERVICE
    ) as AccessibilityManager
    
    // 发送无障碍事件
    fun sendAccessibilityEvent(text: String) {
        if (accessibilityManager.isEnabled) {
            val event = AccessibilityEvent.obtain(
                AccessibilityEvent.TYPE_ANNOUNCEMENT
            ).apply {
                this.text.add(text)
                packageName = context.packageName
                className = "KeyboardScreen"
            }
            accessibilityManager.sendAccessibilityEvent(event)
        }
    }
    
    CompositionLocalProvider(
        LocalAccessibilityFeedback provides ::sendAccessibilityEvent
    ) {
        content()
    }
}
